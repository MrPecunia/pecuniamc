---------------- server.js ----------------
'use strict';

const http = require('http');
const WebSocket = require('ws');

const PORT = process.env.PORT || 10000;
const MAX_PROFILES = 5000;

/** uuid -> { uuid, rank, nick, cape_url } */
const profiles = new Map();

function sanitize(value, max) {
  if (typeof value !== 'string') return '';
  return value.trim().slice(0, max);
}

function isUuid(value) {
  return typeof value === 'string'
    && /^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$/.test(value);
}

function payload(profile) {
  return JSON.stringify({
    type: 'update_profile',
    uuid: profile.uuid,
    rank: profile.rank,
    nick: profile.nick,
    cape_url: profile.cape_url,
  });
}

function broadcast(data, except) {
  for (const client of wss.clients) {
    if (client !== except && client.readyState === WebSocket.OPEN) {
      try { client.send(data); } catch (e) { /* ignore */ }
    }
  }
}

function handleMessage(ws, raw) {
  let msg;
  try {
    msg = JSON.parse(raw);
  } catch (e) {
    return;
  }
  if (!msg || typeof msg !== 'object') return;

  if (msg.type === 'update_profile' || msg.type === 'profile') {
    const uuid = typeof msg.uuid === 'string' ? msg.uuid : '';
    if (!isUuid(uuid)) return;
    const profile = {
      uuid: uuid,
      rank: sanitize(msg.rank, 32),
      nick: sanitize(msg.nick, 16),
      cape_url: sanitize(msg.cape_url || msg.cape, 512),
    };
    profiles.set(uuid, profile);
    if (profiles.size > MAX_PROFILES) {
      profiles.delete(profiles.keys().next().value);
    }
    broadcast(payload(profile), ws);
  }
}

const server = http.createServer((req, res) => {
  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ ok: true, service: 'pecunia-server', clients: wss.clients.size }));
});

const wss = new WebSocket.Server({ server, maxPayload: 4096 });

wss.on('connection', (ws) => {
  ws.isAlive = true;
  ws.on('pong', () => { ws.isAlive = true; });

  // Snapshot: il client appena connesso vede subito tutti i profili.
  for (const profile of profiles.values()) {
    if (ws.readyState === WebSocket.OPEN) {
      try { ws.send(payload(profile)); } catch (e) { /* ignore */ }
    }
  }

  ws.on('message', (data) => handleMessage(ws, data));
});

// Keepalive: chiude i client morti (i client vivi rispondono pong da soli).
setInterval(() => {
  for (const ws of wss.clients) {
    if (ws.isAlive === false) {
      try { ws.terminate(); } catch (e) { /* ignore */ }
      continue;
    }
    ws.isAlive = false;
    try { ws.ping(() => {}); } catch (e) { /* ignore */ }
  }
}, 30000);

server.listen(PORT, () => {
  console.log('pecunia-server listening on port ' + PORT);
});
